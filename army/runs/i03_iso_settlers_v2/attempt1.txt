// --- HELPER FUNCTIONS FOR PROCEDURAL PIXEL ART ---
// Villager drawing helper with exact dimensions, hue-shifted cloth ramps, 3/4 isometric views,
// proper 1px body bobs, opposite arm/leg swings, head with hair, and boots.

function drawVillager(c, role, dir, frame) {
  // c is canvas context. Size: 12x18.
  // 3/4 views: d = down-right (front), u = back, r = right side, l = left side (flipped)
  // frame: 0 = neutral, 1 = walk left leg fwd, 2 = walk right leg fwd
  
  // Map role to cloth color keys
  let cBase = "b", cMid = "B", cDark = "C", cDeep = "d"
  if (role === "builder") { cBase = "o"; cMid = "O"; cDark = "P"; cDeep = "q" }
  if (role === "worker")  { cBase = "g"; cMid = "G"; cDark = "H"; cDeep = "I" }
  if (role === "porter")  { cBase = "n"; cMid = "N"; cDark = "m"; cDeep = "M" }

  // 1px body bob on frames 1 and 2
  let bob = (frame === 1 || frame === 2) ? 1 : 0
  let yHead = 2 + bob
  let yBody = 6 + bob
  let yLegs = 12 + bob

  // --- HEAD & HAIR ---
  // Head is 6x5 ellipse/box, hair on top/back depending on dir
  c.rect(4, yHead, 4, 4, "s") // face base
  c.px(4, yHead + 1, "S") // skin shadow
  
  if (dir === "u") {
    // Back view: full hair
    c.rect(3, yHead - 1, 6, 3, "K")
  } else if (dir === "d") {
    // Front view: hair fringe top
    c.rect(3, yHead - 1, 6, 2, "K")
    c.px(3, yHead + 1, "K")
    c.px(8, yHead + 1, "K")
  } else {
    // Side view r
    c.rect(4, yHead - 1, 4, 2, "K")
    c.px(3, yHead, "K")
  }

  // --- ROLE PROPS / HATS ---
  if (role === "builder") {
    // Yellow hard hat
    c.rect(3, yHead - 2, 6, 2, "y")
    c.px(2, yHead - 1, "Y")
    c.px(9, yHead - 1, "Y")
  } else if (role === "worker") {
    // Straw hat
    c.rect(2, yHead - 2, 8, 1, "y")
    c.rect(3, yHead - 3, 6, 1, "Y")
  } else if (role === "porter" && dir !== "u") {
    // Small backpack visible over shoulder
    c.rect(8, yHead + 2, 2, 4, "M")
  } else if (role === "porter" && dir === "u") {
    c.rect(3, yHead + 2, 6, 5, "M")
  }

  // --- TUNIC (BODY) ---
  // 4-tone cloth ramp shading (top-left light source)
  c.rect(3, yBody, 6, 6, cMid)
  c.rect(3, yBody, 5, 1, cBase) // top lit edge
  c.rect(3, yBody, 1, 5, cBase) // left lit edge
  c.rect(8, yBody + 1, 1, 5, cDark) // right shadow
  c.rect(4, yBody + 5, 5, 1, cDeep) // bottom shadow

  // Belt / tunic hem
  c.rect(3, yBody + 5, 6, 1, "k")

  // --- LEGS & BOOTS ---
  let legL = 4, legR = 7
  if (frame === 1) { legL = 3; legR = 8 }
  if (frame === 2) { legL = 5; legR = 6 }

  c.line(legL, yLegs, legL, yLegs + 3, "K", 1) // left leg + boot
  c.line(legR, yLegs, legR, yLegs + 3, "K", 1) // right leg + boot

  // --- ARMS & TOOLS / GOODS ---
  if (dir === "r" || dir === "d") {
    // Right arm swing / tool
    if (frame === 1) {
      c.px(9, yBody + 2, "S")
      c.px(10, yBody + 3, "S")
    } else if (frame === 2) {
      c.px(8, yBody + 3, "S")
      c.px(9, yBody + 4, "S")
    } else {
      c.px(9, yBody + 3, "S")
    }

    // Role specific hand item in right hand
    if (role === "builder") {
      c.px(10, yBody + 2, "q") // hammer head
      c.px(10, yBody + 3, "K") // hammer handle
    } else if (role === "worker") {
      c.px(10, yBody + 3, "c") // small tool
    }
  } else if (dir === "u") {
    c.px(2, yBody + 3, "S")
    c.px(9, yBody + 3, "S")
  }
}

// --- GENERATE ALL 48 VILLAGER SPRITES (4 roles x 4 directions x 3 frames) ---
const roles = ["idle", "builder", "worker", "porter"]
const dirs = ["r", "d", "u", "l"]

for (const role of roles) {
  for (const dir of dirs) {
    for (let f = 0; f < 3; f++) {
      const name = `settler_${role}_${dir}_${f}`
      if (dir === "l") {
        // Left is mirrored right
        sprite(name, { base: `settler_${role}_r_${f}`, flip: "x" })
      } else {
        sprite(name, { w: 12, h: 18, outline: "k" }, (c) => {
          drawVillager(c, role, dir, f)
        })
      }
    }
  }
}
