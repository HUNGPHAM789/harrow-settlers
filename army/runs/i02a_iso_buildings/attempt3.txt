// --- HELPER FUNCTIONS FOR ISOMETRIC STRUCTURES ---
// Helper to draw an isometric box building with top-left light (left face darker, right face mid, roof lightest)
// Footprint diamond base width = (w_tiles + h_tiles) * 16, height = (w_tiles + h_tiles) * 8
function isoBox(c, x, y, w, h, wallH, leftCol, rightCol, topCol, outlineCol) {
  // x, y is top vertex of the roof / structure
  // Left wall polygon, Right wall polygon, Roof polygon
  c.poly([x, y + wallH, x - w, y + wallH - w/2, x - w, y - w/2, x, y], leftCol)
  c.poly([x, y + wallH, x + h, y + wallH + h/2, x + h, y + h/2, x, y], rightCol)
  c.poly([x, y, x - w, y - w/2, x - w + h, y - w/2 - h/2, x + h, y + h/2], topCol)
  if (outlineCol) {
    c.line(x, y + wallH, x - w, y + wallH - w/2, outlineCol, 1)
    c.line(x, y + wallH, x + h, y + wallH + h/2, outlineCol, 1)
    c.line(x, y + wallH, x, y, outlineCol, 1)
    c.line(x, y - w/2, x - w, y - w/2, outlineCol, 1)
    c.line(x, y + h/2, x + h, y + h/2, outlineCol, 1)
  }
}

function isoRoof(c, x, y, w, h, ridgeH, roofCol, outlineCol) {
  // Isometric pyramid / ridge roof
  c.poly([x, y - ridgeH, x - w, y - w/2, x - w + h, y - w/2 - h/2, x + h, y + h/2], roofCol)
  if (outlineCol) {
    c.line(x, y - ridgeH, x - w, y - w/2, outlineCol, 1)
    c.line(x, y - ridgeH, x + h, y + h/2, outlineCol, 1)
    c.line(x, y - ridgeH, x - w + h, y - w/2 - h/2, outlineCol, 1)
  }
}

function drawScaffoldFrame(c, w, h, width, height) {
  // Wooden scaffolding poles, planks and ladder
  c.rect(0, 0, width, height, "_")
  // Vertical posts
  c.line(4, height - 8, 4, 12, "h", 1)
  c.line(width - 5, height - 8, width - 5, 12, "h", 1)
  c.line(width/2, height - 4, width/2, 8, "i", 1)
  // Horizontal crossbeams
  c.hline(4, height - 20, width - 8, "i")
  c.hline(4, height - 36, width - 8, "i")
  // Ladder rungs
  for (let ly = 16; ly < height - 12; ly += 6) {
    c.hline(width/2 - 3, ly, 6, "j")
  }
  // Partial plank walls
  c.rect(8, height - 28, 12, 10, "i")
  c.rect(width - 20, height - 24, 10, 8, "j")
}
