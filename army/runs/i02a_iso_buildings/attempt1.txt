// --- 1. b_iso_hq (96x104, 3x3 tiles footprint) ---
sprite("b_iso_hq", { w: 96, h: 104, outline: "wd0" }, (c) => {
  // Base footprint diamond center bottom at (48, 104)
  // Left wall polygon
  c.poly([48, 104, 0, 80, 0, 56, 48, 80], "wd1");
  // Right wall polygon
  c.poly([48, 104, 96, 80, 96, 56, 48, 80], "wd2");
  // Main Keep Stone Walls
  c.poly([16, 72, 48, 88, 48, 40, 16, 24], "st1"); // Left face
  c.poly([48, 88, 80, 72, 80, 24, 48, 40], "st2"); // Right face
  // Stone coursing lines
  c.line(20, 68, 44, 80, "st0", 1);
  c.line(24, 60, 44, 70, "st0", 1);
  c.line(52, 70, 72, 60, "st0", 1);
  c.line(52, 80, 76, 68, "st0", 1);

  // Towers / Battlements
  c.rect(12, 16, 12, 24, "st3");
  c.rect(72, 16, 12, 24, "st3");
  // Roof peaks
  c.poly([48, 8, 12, 24, 48, 40], "rf2");
  c.poly([48, 8, 48, 40, 84, 24], "rf3");

  // Wooden Gate & Door
  c.poly([40, 84, 56, 92, 56, 76, 40, 68], "wd3");
  c.rect(44, 74, 8, 14, "wd0");

  // Windows & Portcullis details
  c.rect(26, 32, 6, 8, "st0");
  c.rect(64, 32, 6, 8, "st0");

  // Flag on HQ tower
  c.line(48, 8, 48, 0, "ir2", 1);
  c.poly([48, 0, 58, 4, 48, 8], "cl2");

  // Props: shields / gold trim
  c.rect(44, 50, 8, 6, "gl2");
});

sprite("b_iso_hq_scaffold", { w: 96, h: 104, outline: "sc0" }, (c) => {
  // Scaffold version: wooden poles, partial walls, ladder
  c.poly([16, 72, 48, 88, 48, 40, 16, 24], "wd1");
  c.poly([48, 88, 80, 72, 80, 24, 48, 40], "wd2");
  // Wooden scaffold poles overlay
  for (let x = 12; x <= 84; x += 18) {
    c.line(x, 96, x, 16, "sc2", 2);
  }
  c.hline(12, 40, 72, "sc3", 2);
  c.hline(12, 70, 72, "sc3", 2);
  // Ladder
  c.line(30, 96, 30, 40, "sc3", 1);
  c.line(34, 96, 34, 40, "sc3", 1);
  for (let y = 45; y < 96; y += 6) {
    c.hline(30, y, 4, "sc1", 1);
  }
  // Partial unfinished roof
  c.poly([48, 12, 20, 28, 48, 36], "sc1");
});
