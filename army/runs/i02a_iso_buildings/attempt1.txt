// --- HELPER FUNCTIONS ---
// Must be defined once at top and reused.
// isoBox: draws isometric building body given width, left wall color, right wall color, floor/base
function isoBox(c, w, h, leftColor, rightColor, topColor) {
  const hw = Math.floor(w / 2);
  const baseH = h;
  // Left wall polygon (from bottom-centre up to roof ridge/eaves)
  // Footprint diamond bottom vertex is at (hw, totalH), but here relative to c.w, c.h.
  // We pass canvas dimensions c.w, c.h
  // Left face
  c.poly([hw, c.h, 0, c.h - hw / 2, 0, c.h - hw / 2 - baseH, hw, c.h - hw - baseH], leftColor);
  // Right face
  c.poly([hw, c.h, w, c.h - hw / 2, w, c.h - hw / 2 - baseH, hw, c.h - hw - baseH], rightColor);
}

// isoRoof: draws a 2:1 slope isometric pyramid or ridge roof
function isoRoof(c, w, h, leftRoofColor, rightRoofColor) {
  const hw = Math.floor(w / 2);
  // Ridge from top-left/right to center peak
  c.poly([hw, c.h - w, 0, c.h - hw / 2, hw, c.h - hw - h], leftRoofColor);
  c.poly([hw, c.h - w, w, c.h - hw / 2, hw, c.h - hw - h], rightRoofColor);
}

// Scaffold helper: draws timber framework over building footprint
function isoScaffold(c, w, h) {
  const hw = Math.floor(w / 2);
  // Vertical posts
  c.line(hw, c.h, hw, c.h - h - 16, "h", 1);
  c.line(4, c.h - 8, 4, c.h - h, "h", 1);
  c.line(w - 4, c.h - 8, w - 4, c.h - h, "h", 1);
  // Cross beams along 2:1 slopes
  c.line(2, c.h - hw / 2, hw, c.h - hw, "i", 1);
  c.line(hw, c.h - hw, w - 2, c.h - hw / 2, "i", 1);
  c.line(2, c.h - hw / 2 - 16, hw, c.h - hw - 16, "i", 1);
  c.line(hw, c.h - hw - 16, w - 2, c.h - hw / 2 - 16, "i", 1);
  // Ladder
  c.line(hw - 3, c.h - 4, hw - 3, c.h - h + 8, "j", 1);
  c.line(hw + 3, c.h - 4, hw + 3, c.h - h + 8, "j", 1);
  for(let y = c.h - 8; y > c.h - h + 12; y -= 6) {
    c.line(hw - 3, y, hw + 3, y, "h", 1);
  }
}

// --- 1. b_iso_hq (96x104) ---
// Castle keep with towers, gate, red flags, stone blocks
sprite("b_iso_hq", { w: 96, h: 104, outline: "a" }, (c) => {
  // Base stone footprint 96x48 diamond projection
  isoBox(c, 96, 56, "c", "d", "e");
  // Stone brick texture lines along 2:1 slope
  for(let y = 60; y < 96; y += 8) {
    c.line(16, y, 80, y - 32, "b", 1);
  }
  // Keep upper tower walls
  c.poly([48, 20, 16, 36, 16, 76, 48, 92], "c");
  c.poly([48, 20, 80, 36, 80, 76, 48, 92], "d");
  // Roofs (blue slate)
  isoRoof(c, 96, 28, "q", "r");
  // Central golden banner / flag
  c.line(48, 20, 48, 4, "a", 1);
  c.poly([48, 4, 60, 9, 48, 14], "x");
  // Gate & windows (small polys)
  c.poly([42, 84, 54, 84, 54, 96, 42, 96], "a");
  c.poly([28, 55, 34, 52, 34, 60, 28, 63], "a");
  c.poly([62, 55, 68, 52, 68, 60, 62, 63], "a");
})

sprite("b_iso_hq_scaffold", { w: 96, h: 104, outline: "a" }, (c) => {
  isoBox(c, 96, 56, "h", "i", "j");
  isoScaffold(c, 96, 80);
})
