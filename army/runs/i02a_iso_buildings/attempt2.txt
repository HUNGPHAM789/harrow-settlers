// Helper: draw an isometric box (left face, right face, top/roof)
function isoBox(c, x, y, w, h, height, lColor, rColor, tColor) {
  // w, h are footprint dimensions in isometric tile steps (16x8 pixels per step)
  // left wall: from (x, y + h*8) down by height to (x + w*16, y + w*8 + h*8)
  const leftX = x;
  const leftY = y + h * 8;
  const midX = x + w * 16;
  const midY = y + (w + h) * 8;
  const rightX = x + (w + h) * 16;
  const rightY = y + h * 8; // wait, correct iso geometry:
  // Footprint diamond for w x h:
  // Top: ( (w+h)*16, y )
  // Right: ( (w+h)*32, y + (w+h)*8 ) -> wait, footprint width is (w+h)*16 total.
  // Let's use precise iso box projection:
  // Center bottom of footprint is at x, y + (w+h)*8.
  // Left face polygon:
  c.poly([
    x, y + w * 8,                     // top-left ridge/corner
    x + w * 16, y + (w + h) * 8,      // bottom vertex
    x + w * 16, y + (w + h) * 8 + height, // bottom vertex down
    x, y + w * 8 + height             // top-left down
  ], lColor);

  // Right face polygon:
  c.poly([
    x + w * 16, y + (w + h) * 8,      // bottom vertex
    x + (w + h) * 16, y + h * 8,      // top-right corner
    x + (w + h) * 16, y + h * 8 + height, // top-right down
    x + w * 16, y + (w + h) * 8 + height  // bottom vertex down
  ], rColor);

  // Top cap (flat box top):
  c.poly([
    x, y + w * 8,                     // left
    x + w * 16, y,                    // top
    x + (w + h) * 16, y + h * 8,      // right
    x + w * 16, y + (w + h) * 8       // bottom
  ], tColor);
}

// Helper: draw an isometric roof (pyramid or ridge)
function isoRoof(c, x, y, w, h, height, rLeft, rRight, rTop) {
  const topX = x + w * 16;
  const topY = y;
  const leftX = x;
  const leftY = y + w * 8;
  const rightX = x + (w + h) * 16;
  const rightY = y + h * 8;
  const botX = x + w * 16;
  const botY = y + (w + h) * 8;
  const peakY = y - height;

  // Left roof slope
  c.poly([leftX, leftY, topX, topY, topX, peakY, leftX, leftY - height * 0.7], rLeft);
  // Right roof slope
  c.poly([topX, topY, rightX, rightY, rightX, rightY - height * 0.7, topX, peakY], rRight);
  // Front/Back gables or highlights if needed
  c.poly([topX, topY, topX, peakY, rightX, rightY - height * 0.7], rTop);
}

// Scaffold generator for incomplete buildings
function drawScaffold(c, w, h, totalW, totalH) {
  // Wooden pole vertical posts and horizontal crossbeams following isometric grid
  const cols = Math.max(2, w);
  const rows = Math.max(2, h);
  for (let i = 0; i <= cols; i++) {
    const px = (i / cols) * totalW * 0.5 + (cols - i) * 8;
    const py = (i / cols) * totalH * 0.25 + i * 8;
    c.line(px, py + 20, px, totalH - 10, "P1", 2);
    c.line(px, py + 20, px, totalH - 10, "P2", 1);
  }
  // Cross braces
  c.line(4, totalH * 0.5, totalW - 4, totalH * 0.3, "P2", 2);
  c.line(4, totalH * 0.7, totalW - 4, totalH * 0.5, "P2", 2);
  // Ladder on front
  const lx = totalW * 0.5 - 4;
  c.vline(lx, totalH * 0.3, totalH * 0.5, "P3");
  c.vline(lx + 6, totalH * 0.3, totalH * 0.5, "P3");
  for (let ly = totalH * 0.35; ly < totalH * 0.75; ly += 6) {
    c.hline(lx, ly, 6, "P2");
  }
}

// 1. b_iso_hq (96x104) - 3x3 tiles footprint
// -----------------------------------------------------------------------------
sprite("b_iso_hq", { w: 96, h: 104, outline: "K" }, (c) => {
  // Footprint: 3x3 tiles -> 96px wide, 48px base diamond
  // Castle keep: stone walls, towers, blue roof, grand wooden doors, golden flag
  const ox = 0;
  const oy = 16;
  // Base walls (S2 left, S3 right, S4 top)
  isoBox(c, ox, oy + 8, 3, 3, 40, "S2", "S3", "S4");
  
  // Stone course lines (2:1 slope)
  for (let dy = 12; dy < 38; dy += 8) {
    c.line(ox + 4, oy + 12 + dy, ox + 24, oy + 24 + dy, "S1", 1);
    c.line(ox + 24, oy + 24 + dy, ox + 48, oy + 12 + dy, "S2", 1);
  }

  // Central keep tower / second tier
  isoBox(c, ox + 24, oy - 12, 1.5, 1.5, 30, "S3", "S4", "S5");

  // Roofs (blue/red conical or pyramidal roofs)
  isoRoof(c, ox + 24, oy - 12, 1.5, 1.5, 24, "R1", "R2", "R3");

  // Grand doors & arched windows
  c.poly([ox + 40, oy + 32, ox + 48, oy + 36, ox + 48, oy + 50, ox + 40, oy + 46], "W1");
  c.poly([ox + 48, oy + 36, ox + 56, oy + 32, ox + 56, oy + 46, ox + 48, oy + 50], "W2");
  c.rect(ox + 45, oy + 38, 6, 12, "W3");

  // Windows
  c.rect(ox + 20, oy + 16, 4, 6, "G2");
  c.rect(ox + 68, oy + 16, 4, 6, "G2");
  c.rect(ox + 46, oy - 4, 4, 6, "G2");

  // Flag pole and banner
  c.line(ox + 48, oy - 36, ox + 48, oy - 12, "M2", 2);
  c.poly([ox + 48, oy - 36, ox + 60, oy - 30, ox + 48, oy - 24], "B2");
  c.rect(ox + 47, oy - 38, 3, 3, "B3"); // gold finial
});

sprite("b_iso_hq_scaffold", { base: "b_iso_hq" }, (c) => {
  drawScaffold(c, 3, 3, 96, 104);
});
