// Helper: draw an isometric box (left wall, right wall, roof) with 2:1 slope edges
function isoBox(c, w, h, roofH, leftColor, rightColor, roofColor) {
  // c is canvas. w, h are footprint dimensions in pixels, roofH is roof height.
  // Footprint bottom-centre is at (w/2, h + roofH) approximately depending on projection.
  // We use poly for clean 2:1 isometric faces.
  const hw = w / 2;
  const hh = h / 2;
  
  // Left wall polygon
  c.poly([hw, hh + roofH, 0, hh * 0.5 + roofH, 0, hh * 0.5 + roofH - h, hw, hh + roofH - h], leftColor);
  // Right wall polygon
  c.poly([hw, hh + roofH, w, hh * 0.5 + roofH, w, hh * 0.5 + roofH - h, hw, hh + roofH - h], rightColor);
  // Roof pyramid / top
  c.poly([hw, roofH - h - roofH, w, hh * 0.5 + roofH - h, hw, hh + roofH - h, 0, hh * 0.5 + roofH - h], roofColor);
}

// Helper: draw timber scaffold over building bounds
function isoScaffold(c, w, h, totalH) {
  // Wooden poles and cross braces
  const step = 16;
  for (let x = 8; x < w - 4; x += step) {
    c.line(x, totalH - 4, x, 10, "c", 1);
    c.line(x - 4, totalH - 12, x + 4, totalH - 20, "b", 1);
    c.line(x - 4, totalH - 28, x + 4, totalH - 36, "b", 1);
  }
  // Horizontal beams & ladder
  c.hline(4, totalH - 16, w - 8, "d");
  c.hline(4, totalH - 32, w - 8, "d");
  // Ladder on left side
  c.vline(12, 16, totalH - 24, "e");
  c.vline(16, 16, totalH - 24, "e");
  c.hline(12, 24, 4, "c");
  c.hline(12, 36, 4, "c");
  c.hline(12, 48, 4, "c");
}
