import { GOODS, BUILDINGS, BUILD_ORDER } from '../sim/defs.js';

/**
 * UI Class manages the DOM-based HUD overlay.
 * Handles user input for building, road painting, and game speed.
 * Uses dirty-checking to minimize DOM thrashing.
 */
export class UI {
  constructor(root, game, renderer, callbacks) {
    this.root = root;
    this.game = game;
    this.renderer = renderer;
    this.callbacks = callbacks;

    this.tool = null; // null | 'road' | 'demolish' | {build: type}
    this._mouse = { x: 0, y: 0, tx: 0, ty: 0 };
    this._lastState = {
      stock: {},
      pop: -1,
      housing: -1,
      foodRate: -1,
      time: -1,
      speed: -1,
      selectedId: null,
      selectedState: '',
      selectedProgress: -1,
      victory: false,
      objectives: []
    };

    this._elements = {};
    this._initDOM();
    this._setupListeners();
  }

  /**
   * Main update loop called every frame.
   */
  update() {
    this._updateToasts();
    this._updateTopBar();
    this._updateBuildPalette();
    this._updateInfoPanel();
    this._updateObjectives();
    this._updateModals();
    this._updateMinimap();
    this._updatePlacementTooltip();
  }

  destroy() {
    window.removeEventListener('keydown', this._onKeyDown);
    this.root.textContent = '';
  }

  handleCanvasClick(tx, ty, button) {
    if (this.tool === 'road') {
      this.game.paintRoad(tx, ty, button === 0);
      return true;
    }
    if (this.tool === 'demolish') {
      const b = this.game.buildings.find(b => 
        tx >= b.x && tx < b.x + BUILDINGS[b.type].size[0] &&
        ty >= b.y && ty < b.y + BUILDINGS[b.type].size[1]
      );
      if (b && b.type !== 'hq') this.game.demolish(b.id);
      return true;
    }
    if (this.tool && this.tool.build) {
      const res = this.game.place(this.tool.build, tx, ty);
      if (res.ok) {
        if (!window.event?.shiftKey) this.tool = null;
      }
      return true;
    }

    // Default: Selection
    const b = this.game.buildings.find(b => 
      tx >= b.x && tx < b.x + BUILDINGS[b.type].size[0] &&
      ty >= b.y && ty < b.y + BUILDINGS[b.type].size[1]
    );
    this.game.selectedBuildingId = b ? b.id : null;
    return !!b;
  }
