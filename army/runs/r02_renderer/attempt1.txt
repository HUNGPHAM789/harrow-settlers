import { T, BUILDINGS, TILE } from '../sim/defs.js';
import { idx, inBounds } from '../sim/world.js';
import { terrainSprite, settlerSprite } from './sprites.js';

/**
 * Renderer handles the visual representation of the game world.
 * Uses Canvas2D with chunked terrain caching and Y-sorted entity drawing.
 */
export class Renderer {
  constructor(canvas, game, sprites) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d', { alpha: false });
    this.game = game;
    this.sprites = sprites;

    // Camera state
    this.camX = game.world.start.x * TILE;
    this.camY = game.world.start.y * TILE;
    this.zoom = 2;
    this.targetZoom = 2;

    // Interaction state
    this.mouse = { x: 0, y: 0, down: false, lastX: 0, lastY: 0 };
    this.keys = {};
    this.ghostType = null;
    this.selectedId = null;

    // Particle pool
    this.particles = Array.from({ length: 128 }, () => ({
      active: false, x: 0, y: 0, vx: 0, vy: 0, life: 0, maxLife: 0
    }));
    this.smokeTimer = 0;

    // Render pool for Y-sorting (pre-allocated to avoid GC)
    this.renderPool = Array.from({ length: 4000 }, () => ({
      y: 0, type: 0, name: '', x: 0, y2: 0, w: 0, h: 0, val: 0, opts: {}
    }));
    this.renderCount = 0;

    // Chunk cache (16x16 tiles per chunk)
    this.chunkSize = 16;
    this.chunksW = Math.ceil(game.world.w / this.chunkSize);
    this.chunksH = Math.ceil(game.world.h / this.chunkSize);
    this.chunks = [];
    this.dirtyChunks = new Set();
    this._initChunks();

    // Minimap cache
    this.minimapTerrain = document.createElement('canvas');
    this.minimapTerrain.width = game.world.w;
    this.minimapTerrain.height = game.world.h;
    this._drawMinimapTerrain();

    // Setup events
    this._setupEvents();
    this.resize();
  }

  _initChunks() {
    for (let cy = 0; cy < this.chunksH; cy++) {
      for (let cx = 0; cx < this.chunksW; cx++) {
        const off = document.createElement('canvas');
        off.width = this.chunkSize * TILE;
        off.height = this.chunkSize * TILE;
        this.chunks.push({ canvas: off, ctx: off.getContext('2d') });
        this.dirtyChunks.add(cy * this.chunksW + cx);
      }
    }
  }

  _setupEvents() {
    this.canvas.addEventListener('mousedown', e => {
      this.mouse.down = true;
      this.mouse.lastX = e.clientX;
      this.mouse.lastY = e.clientY;
    });
    window.addEventListener('mousemove', e => {
      const rect = this.canvas.getBoundingClientRect();
      this.mouse.x = e.clientX - rect.left;
      this.mouse.y = e.clientY - rect.top;
      if (this.mouse.down) {
        const dx = e.clientX - this.mouse.lastX;
        const dy = e.clientY - this.mouse.lastY;
        this.camX -= dx / this.zoom;
        this.camY -= dy / this.zoom;
        this.mouse.lastX = e.clientX;
        this.mouse.lastY = e.clientY;
      }
    });
    window.addEventListener('mouseup', () => this.mouse.down = false);
    this.canvas.addEventListener('wheel', e => {
      e.preventDefault();
      if (e.deltaY < 0) this.targetZoom = Math.min(4, this.targetZoom + 1);
      else if (e.deltaY > 0) this.targetZoom = Math.max(1, this.targetZoom - 1);
    }, { passive: false });
    window.addEventListener('keydown', e => this.keys[e.code] = true);
    window.addEventListener('keyup', e => this.keys[e.code] = false);
  }

  resize() {
    const dpr = window.devicePixelRatio || 1;
    const rect = this.canvas.getBoundingClientRect();
    this.canvas.width = rect.width * dpr;
    this.canvas.height = rect.height * dpr;
  }

  setGhost(type) { this.ghostType = type; }
  setSelected(id) { this.selectedId = id; }

  screenToTile(sx, sy) {
    const dpr = window.devicePixelRatio || 1;
    const worldX = (sx * dpr - this.canvas.width / 2) / this.zoom + this.camX;
    const worldY = (sy * dpr - this.canvas.height / 2) / this.zoom + this.camY;
    return { x: Math.floor(worldX / TILE), y: Math.floor(worldY / TILE) };
  }

  tileToScreen(tx, ty) {
    const dpr = window.devicePixelRatio || 1;
    const sx = ((tx * TILE - this.camX) * this.zoom + this.canvas.width / 2) / dpr;
    const sy = ((ty * TILE - this.camY) * this.zoom + this.canvas.height / 2) / dpr;
    return { x: sx, y: sy };
  }

  render(dt) {
    this._update(dt);
    this._draw();
  }

  _update(dt) {
    // Smooth zoom
    const zoomLerp = 1 - Math.pow(0.001, dt);
    this.zoom += (this.targetZoom - this.zoom) * zoomLerp;

    // Keyboard pan
    const panSpeed = 400 * dt / this.zoom;
    if (this.keys['KeyW'] || this.keys['ArrowUp']) this.camY -= panSpeed;
    if (this.keys['KeyS'] || this.keys['ArrowDown']) this.camY += panSpeed;
    if (this.keys['KeyA'] || this.keys['ArrowLeft']) this.camX -= panSpeed;
    if (this.keys['KeyD'] || this.keys['ArrowRight']) this.camX += panSpeed;

    // Clamp camera
    const margin = 100;
    this.camX = Math.max(-margin, Math.min(this.game.world.w * TILE + margin, this.camX));
    this.camY = Math.max(-margin, Math.min(this.game.world.h * TILE + margin, this.camY));

    // Update particles
    for (let i = 0; i < this.particles.length; i++) {
      const p = this.particles[i];
      if (!p.active) continue;
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      p.life -= dt;
      if (p.life <= 0) p.active = false;
    }

    // Spawn smoke
    this.smokeTimer += dt;
    if (this.smokeTimer > 0.3) {
      this.smokeTimer = 0;
      for (let i = 0; i < this.game.buildings.length; i++) {
        const b = this.game.buildings[i];
        if (b.state === 'active' && b.working) {
          this._spawnSmoke(b.x * TILE + TILE, b.y * TILE);
        }
      }
    }
  }

  _spawnSmoke(x, y) {
    for (let i = 0; i < this.particles.length; i++) {
      const p = this.particles[i];
      if (!p.active) {
        p.active = true;
        p.x = x + (Math.random() - 0.5) * 4;
        p.y = y;
        p.vx = (Math.random() - 0.5) * 10;
        p.vy = -20 - Math.random() * 20;
        p.maxLife = 1.0 + Math.random();
        p.life = p.maxLife;
        break;
      }
    }
  }

  _draw() {
    const { ctx, canvas, zoom, camX, camY, game } = this;
    const dpr = window.devicePixelRatio || 1;

    ctx.fillStyle = '#001122';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.save();
    ctx.translate(canvas.width / 2, canvas.height / 2);
    ctx.scale(zoom, zoom);
    ctx.translate(-camX, -camY);

    this._drawTerrain();
    this._drawRoads();
    this._drawEntities();
    this._drawTerritoryBorders();
    this._drawOverlays();

    ctx.restore();

    this._drawDayNight();
  }

          this._
