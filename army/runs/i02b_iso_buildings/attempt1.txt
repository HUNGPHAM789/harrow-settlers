// --- REUSABLE HELPER FUNCTIONS ---
// Draws a true isometric box with correct 2:1 slope faces, left wall darker, right wall mid, roof lightest.
// Footprint is based on tile width/height.
function isoBox(c, x, y, wW, wH, wallH, leftCol, rightCol, roofTopCol) {
  // Left wall face
  c.poly([x, y + wallH, x + wW, y + wallH + wW / 2, x + wW, y + wallH + wW / 2 + wallH, x, y + wW / 2 + wallH], leftCol)
  // Right wall face
  c.poly([x + wW, y + wallH + wW / 2, x + wW + wH, y + wallH + wW / 2 - wH / 2, x + wW + wH, y + wallH + wW / 2 - wH / 2 + wallH, x + wW, y + wallH + wW / 2 + wallH], rightCol)
  // Roof / Top
  c.poly([x, y + wallH, x + wW, y + wallH + wW / 2, x + wW + wH, y + wallH + wW / 2 - wH / 2, x + wH, y + wallH - wH / 2], roofTopCol)
}

// Draws textured plank or stone course lines following 2:1 slope
function drawCourses(c, x, y, wW, wH, wallH, step, col) {
  for (let h = step; h < wallH; h += step) {
    c.line(x, y + wallH + (wW/2) * (h/wallH) - h, x + wW, y + wallH + wW/2 + (wW/2)*(h/wallH) - h, col, 1)
  }
}

// Standard scaffold structure for unbuilt buildings (_scaffold version)
function drawScaffold(c, w, h, footprintW, footprintH) {
  // Wooden posts and framing beams
  c.rect(0, 0, w, h, "_")
  const cx = w / 2
  const bottomY = h - 4
  
  // Vertical support poles
  c.line(cx - 24, bottomY - 32, cx - 24, bottomY, "b", 2)
  c.line(cx + 24, bottomY - 32, cx + 24, bottomY, "b", 2)
  c.line(cx - 10, bottomY - 44, cx - 10, bottomY, "c", 2)
  c.line(cx + 10, bottomY - 44, cx + 10, bottomY, "c", 2)

  // Horizontal crossbeams & diagonal braces
  c.line(cx - 28, bottomY - 24, cx + 28, bottomY - 24, "c", 2)
  c.line(cx - 28, bottomY - 12, cx + 28, bottomY - 12, "b", 2)
  c.line(cx - 24, bottomY - 28, cx + 10, bottomY - 40, "d", 1)
  c.line(cx + 24, bottomY - 28, cx - 10, bottomY - 40, "d", 1)

  // Partial scaffolding platforms & ladder rungs
  c.rect(cx - 18, bottomY - 26, 36, 3, "d")
  c.line(cx - 6, bottomY - 36, cx - 6, bottomY - 4, "c", 1)
  c.line(cx - 2, bottomY - 36, cx - 2, bottomY - 4, "c", 1)
  for (let ly = bottomY - 34; ly < bottomY - 6; ly += 6) {
    c.hline(cx - 6, ly, 4, "e")
  }
  // Soft contact shadow
  c.ellipse(cx, bottomY, footprintW, 4, "a")
}

  c.poly([37, 59, 43, 62, 43, 67, 37, 64], "b")

  c.line(sx - 26, sy + 26, sx + 26, sy - 26, "b", 2)
