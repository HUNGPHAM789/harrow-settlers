// Helper: draw an isometric box (left wall, right wall, roof)
function isoBox(c, x, y, w, h, z, { left, right, top }) {
  // left wall: from bottom-left to bottom and up
  c.poly([x, y - z, x + w, y - z + (w / 2), x + w, y + (w / 2), x, y], left)
  // right wall: from bottom to bottom-right and up
  c.poly([x + w, y - z + (w / 2), x + w + h, y - z - h + (w / 2) + (h / 2), x + w + h, y + h + (w / 2) - (h / 2), x + w, y + (w / 2)], right)
  // roof top diamond
  c.poly([x + w, y - z - h + (w / 2), x + w + h, y - z - h + (w / 2) + (h / 2), x, y - z - h + h, x - h, y - z - h + (w / 2)], top)
}

// Helper: scaffold version for a building
function drawScaffold(c, width, height, totalH) {
  // Wooden posts and cross beams following 2:1 slope
  const col = "L"
  const darkCol = "k"
  c.line(4, height - 2, 4, height - totalH, col, 1)
  c.line(width - 4, height - 2, width - 4, height - totalH, col, 1)
  c.line(width / 2, height, width / 2, height - totalH, col, 1)
  // Cross bars
  c.hline(4, height - totalH / 2, width - 8, col)
  c.hline(6, height - totalH / 4, width - 12, col)
  // Ladder
  c.vline(width / 2 - 2, height - totalH, totalH, darkCol)
  c.vline(width / 2 + 2, height - totalH, totalH, darkCol)
  for (let py = height - totalH + 4; py < height; py += 6) {
    c.hline(width / 2 - 2, py, 4, "W")
  }
}
