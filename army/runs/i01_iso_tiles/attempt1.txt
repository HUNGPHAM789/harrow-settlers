// Base isometric diamond helper (32x16)
function drawDiamond(c, fillKey, leftKey, rightKey, topKey) {
  // 32 wide, 16 high. Left=0,8; Top=16,0; Right=31,8; Bottom=16,15
  c.poly([16,0, 32,8, 16,16, 0,8], fillKey);
  // Left slope highlight/shadow
  c.line(0, 8, 16, 16, leftKey);
  // Right slope shadow
  c.line(16, 16, 31, 8, rightKey);
  // Top corner light
  c.px(16, 0, topKey);
  c.px(15, 1, topKey);
  c.px(17, 1, topKey);
}
