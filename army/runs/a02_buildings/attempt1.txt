// --- HELPER FUNCTIONS FOR DRAWING ---

// 1px stepped Bresenham line to eliminate broken/floating pixels
function drawLine(c, x0, y0, x1, y1, col) {
  x0 = Math.round(x0); y0 = Math.round(y0);
  x1 = Math.round(x1); y1 = Math.round(y1);
  const dx = Math.abs(x1 - x0);
  const dy = Math.abs(y1 - y0);
  const sx = x0 < x1 ? 1 : -1;
  const sy = y0 < y1 ? 1 : -1;
  let err = dx - dy;
  let x = x0;
  let y = y0;
  while (true) {
    c.px(x, y, col);
    if (x === x1 && y === y1) break;
    const e2 = 2 * err;
    if (e2 > -dy) { err -= dy; x += sx; }
    if (e2 < dx) { err += dx; y += sy; }
  }
}

// Textured Wooden Plank Wall with lighting
function drawPlankWall(c, x, y, w, h) {
  c.rect(x, y, w, h, "t");
  c.box(x, y, w, h, "k");
  for (let py = y + 3; py < y + h - 1; py += 3) {
    c.hline(x + 1, py, w - 2, "w");
    if (py + 1 < y + h - 1) {
      c.hline(x + 1, py + 1, w - 2, "T");
    }
  }
  for (let py = y + 1; py < y + h - 1; py += 3) {
    let offset = Math.floor((py - y) / 3) % 2 === 0 ? 5 : 10;
    for (let px = x + offset; px < x + w - 2; px += 10) {
      c.px(px, py, "W");
    }
  }
  c.vline(x + 1, y + 1, h - 2, "l");
  c.vline(x + w - 2, y + 1, h - 2, "W");
}

// Coursed Stone Wall with quoins and mortar lines
function drawStoneWall(c, x, y, w, h) {
  c.rect(x, y, w, h, "a");
  c.box(x, y, w, h, "k");
  for (let sy = y + 4; sy < y + h - 1; sy += 4) {
    c.hline(x + 1, sy, w - 2, "s");
  }
  let row = 0;
  for (let sy = y + 1; sy < y + h - 1; sy += 4) {
    let shift = (row % 2) * 4 + 3;
    row++;
    for (let sx = x + shift; sx < x + w - 2; sx += 8) {
      c.vline(sx, sy, Math.min(3, y + h - 1 - sy), "s");
    }
  }
  c.vline(x + 1, y + 1, h - 2, "A");
  c.hline(x + 1, y + 1, w - 2, "A");
  c.vline(x + w - 2, y + 1, h - 2, "S");
  c.hline(x + 1, y + h - 2, w - 2, "S");
}

// Gable Roof - Clay Tile (Pitch-scaled row by row to fit roof edge perfectly)
function drawGableClayRoof(c, apexX, apexY, leftX, rightX, bottomY) {
  const h = bottomY - apexY;
  if (h <= 0) return;
  for (let y = apexY; y <= bottomY; y++) {
    const progress = (y - apexY) / h;
    const xl = Math.floor(apexX - progress * (apexX - leftX));
    const xr = Math.ceil(apexX + progress * (rightX - apexX));
    for (let x = xl; x <= xr; x++) {
      if (x === xl || x === xr) {
        c.px(x, y, "k");
      } else if (x === xl + 1) {
        c.px(x, y, "y");
      } else if (x === xr - 1) {
        c.px(x, y, "R");
      } else {
        const row = y - apexY;
        if (row % 3 === 0) {
          c.px(x, y, "r");
        } else if (row % 3 === 1) {
          c.px(x, y, x % 3 === 0 ? "C" : "c");
        } else {
          c.px(x, y, x % 3 === 1 ? "C" : "c");
        }
      }
    }
  }
  drawLine(c, leftX - 1, bottomY + 1, rightX + 1, bottomY + 1, "k");
  c.hline(leftX, bottomY, (rightX - leftX) + 1, "r");
}

// Gable Roof - Thatch / Straw (Pitch-scaled row by row)
function drawGableThatchRoof(c, apexX, apexY, leftX, rightX, bottomY) {
  const h = bottomY - apexY;
  if (h <= 0) return;
  for (let y = apexY; y <= bottomY; y++) {
    const progress = (y - apexY) / h;
    const xl = Math.floor(apexX - progress * (apexX - leftX));
    const xr = Math.ceil(apexX + progress * (rightX - apexX));
    for (let x = xl; x <= xr; x++) {
      if (x === xl || x === xr) {
        c.px(x, y, "k");
      } else if (x === xl + 1) {
        c.px(x, y, "e");
      } else if (x === xr - 1) {
        c.px(x, y, "H");
      } else {
        const row = y - apexY;
        if (row % 2 === 0) {
          c.px(x, y, x % 2 === 0 ? "F" : "f");
        } else {
          c.px(x, y, x % 3 === 0 ? "F" : "f");
        }
      }
    }
  }
  drawLine(c, leftX - 1, bottomY + 1, rightX + 1, bottomY + 1, "k");
  c.hline(leftX, bottomY, (rightX - leftX) + 1, "h");
}

// Wooden Ladder
function drawLadder(c, x, y0, y1) {
  c.vline(x, y0, y1 - y0 + 1, "k");
  c.vline(x + 1, y0, y1 - y0 + 1, "w");
  c.vline(x + 3, y0, y1 - y0 + 1, "k");
  c.vline(x + 4, y0, y1 - y0 + 1, "w");
  for (let ry = y0 + 2; ry < y1; ry += 3) {
    c.hline(x + 1, ry, 3, "l");
  }
}

// Scaffold Walkway & Posts
function drawScaffoldPosts(c, posts, y0, y1) {
  for (const px of posts) {
    c.vline(px, y0, y1 - y0 + 1, "k");
    c.vline(px + 1, y0, y1 - y0 + 1, "T");
    c.vline(px + 2, y0, y1 - y0 + 1, "W");
  }
}

function drawScaffoldPlatform(c, x0, x1, y) {
  c.hline(x0, y, x1 - x0 + 1, "k");
  c.hline(x0, y + 1, x1 - x0 + 1, "l");
  c.hline(x0, y + 2, x1 - x0 + 1, "w");
}
