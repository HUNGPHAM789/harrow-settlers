// --- Drawing Helpers ---

function drawStoneBlock(c, x, y, w, h) {
  c.rect(x, y, w, h, "s");
  c.box(x, y, w, h, "k");
  c.hline(x + 1, y + 1, w - 2, "S");
  c.vline(x + 1, y + 1, h - 2, "L");
  c.hline(x + 1, y + h - 1, w - 2, "w");
  c.vline(x + w - 1, y + 1, h - 2, "w");
}

function drawTimberWall(c, x, y, w, h) {
  c.rect(x, y, w, h, "m");
  c.box(x, y, w, h, "k");
  // Vertical plank seams
  for (let px = x + 4; px < x + w - 2; px += 4) {
    c.vline(px, y + 1, h - 2, "B");
    c.vline(px + 1, y + 1, h - 2, "M");
  }
}

function drawPitchedRoof(c, x, y, w, h, darkCol, midCol, lightCol, hiCol) {
  const midX = Math.floor(x + w / 2);
  for (let row = 0; row < h; row++) {
    const curY = y + row;
    const widthAtRow = Math.floor((row / h) * w);
    const x1 = midX - Math.floor(widthAtRow / 2);
    const x2 = midX + Math.floor(widthAtRow / 2);
    if (x2 > x1) {
      c.hline(x1, curY, x2 - x1 + 1, midCol);
      c.px(x1, curY, hiCol);
      c.px(x1 + 1, curY, lightCol);
      c.px(x2, curY, darkCol);
    }
  }
  // Roof outline
  c.line(midX, y, x, y + h, "k");
  c.line(midX, y, x + w - 1, y + h, "k");
  c.hline(x, y + h, w, "k");
}

function drawThatchRoof(c, x, y, w, h) {
  drawPitchedRoof(c, x, y, w, h, "y", "Y", "t", "T");
}

function drawTileRoof(c, x, y, w, h) {
  drawPitchedRoof(c, x, y, w, h, "r", "R", "R", "o");
}

function drawDoor(c, x, y, w, h) {
  c.rect(x, y, w, h, "B");
  c.box(x, y, w, h, "k");
  c.hline(x + 1, y + 1, w - 2, "m");
  c.px(x + w - 2, y + Math.floor(h / 2), "T"); // Iron/brass handle
}

function drawWindow(c, x, y) {
  c.rect(x, y, 4, 5, "a");
  c.box(x, y, 4, 5, "k");
  c.px(x + 1, y + 1, "F");
}

function drawSmoke(c, x, y) {
  c.rect(x - 1, y - 2, 3, 2, "L");
  c.rect(x - 3, y - 5, 4, 3, "S");
  c.rect(x - 5, y - 9, 5, 4, "L");
}

function drawGenericScaffold(c, W, H, propType) {
  // Ground shadow
  c.rect(2, H - 4, W - 4, 3, "k");

  // Partial base wall under construction
  c.rect(4, H - 12, W - 8, 9, "s");
  c.box(4, H - 12, W - 8, 9, "k");
  for (let bx = 8; bx < W - 8; bx += 8) {
    c.vline(bx, H - 12, 9, "w");
  }
  c.hline(4, H - 7, W - 8, "w");

  // Vertical scaffold poles
  const topY = 10;
  const poles = [];
  for (let px = 4; px <= W - 4; px += 12) poles.push(px);
  if (poles[poles.length - 1] < W - 5) poles.push(W - 5);

  for (const px of poles) {
    c.vline(px, topY, H - topY - 3, "k");
    c.vline(px + 1, topY, H - topY - 3, "B");
    c.px(px + 1, topY, "M");
  }

  // Horizontal staging planks
  const levels = [H - 16, H - 28, H - 40];
  if (H >= 64) levels.push(H - 52);

  for (const ly of levels) {
    if (ly >= topY) {
      c.rect(2, ly, W - 4, 3, "k");
      c.rect(3, ly + 1, W - 6, 1, "Y");
      c.rect(3, ly, W - 6, 1, "m");
    }
  }

  // Diagonal wood braces
  for (let i = 0; i < poles.length - 1; i++) {
    const x1 = poles[i] + 1;
    const x2 = poles[i + 1];
    const y1 = levels[0] || H - 16;
    const y2 = levels[1] || topY + 8;
    if (y2 >= topY) {
      c.line(x1, y1, x2, y2, "b");
    }
  }

  // Ladder on right side
  const lx = W - 9;
  c.vline(lx, topY + 4, H - topY - 7, "k");
  c.vline(lx + 4, topY + 4, H - topY - 7, "k");
  c.vline(lx + 1, topY + 4, H - topY - 7, "B");
  c.vline(lx + 3, topY + 4, H - topY - 7, "B");
  for (let ly = topY + 8; ly < H - 6; ly += 5) {
    c.hline(lx + 1, ly, 3, "Y");
  }

  // Specific props per building scaffold type
  if (propType === 'hq') {
    // Castle crane / hoist
    c.rect(6, 4, 14, 3, "B");
    c.box(6, 4, 14, 3, "k");
    c.vline(12, 7, 14, "k"); // Rope
    drawStoneBlock(c, 9, 21, 7, 6);
  } else if (propType === 'woodcutter') {
    // Log stack prop
    c.rect(6, H - 8, 8, 4, "B");
    c.box(6, H - 8, 8, 4, "k");
    c.px(7, H - 6, "m");
  } else if (propType === 'sawmill') {
    // Timber planks
    c.rect(6, H - 8, 12, 4, "y");
    c.box(6, H - 8, 12, 4, "k");
    c.hline(7, H - 6, 10, "Y");
  } else if (propType === 'farm') {
    // Hay bales
    c.rect(6, H - 9, 8, 6, "Y");
    c.box(6, H - 9, 8, 6, "k");
    c.rect(7, H - 8, 6, 4, "t");
  } else if (propType === 'mill') {
    // Windmill sail hub frame
    c.circ(16, 18, 5, "k");
    c.circ(16, 18, 4, "B");
    c.line(16, 18, 4, 6, "k");
    c.line(16, 18, 28, 30, "k");
  } else if (propType === 'smith') {
    // Anvil base
    c.rect(10, H - 8, 8, 5, "B");
    c.box(10, H - 8, 8, 5, "k");
    c.rect(9, H - 11, 10, 3, "i");
    c.box(9, H - 11, 10, 3, "k");
  } else if (propType === 'mine') {
    // Mine tunnel frame draft
    c.rect(8, H - 18, 12, 14, "b");
    c.box(8, H - 18, 12, 14, "k");
    c.rect(10, H - 16, 8, 12, "k");
  } else if (propType === 'market') {
    // Stall counter & crates
    c.rect(6, H - 9, 14, 5, "m");
    c.box(6, H - 9, 14, 5, "k");
  } else if (propType === 'monument') {
    // Monument Great Hall crane
    c.rect(16, 2, 24, 4, "B");
    c.box(16, 2, 24, 4, "k");
    c.vline(28, 6, 20, "k");
    drawStoneBlock(c, 24, 26, 9, 8);
  }
}
