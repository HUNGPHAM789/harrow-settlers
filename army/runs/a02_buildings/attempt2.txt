// Helper for drawing generic wooden scaffolding for any footprint size
function drawScaffold(c, w, h) {
  // Stone footing base
  c.rect(2, h - 6, w - 4, 6, 's');
  c.rect(3, h - 5, w - 6, 2, 'S');
  c.box(2, h - 6, w - 4, 6, 'k');

  // Low partially built wall (stone / timber)
  c.rect(4, h - 16, w - 8, 10, 'S');
  c.rect(5, h - 15, (w - 10) / 2, 8, 'l');
  c.box(4, h - 16, w - 8, 10, 'k');

  // Scaffold vertical poles
  const spacing = 10;
  for (let x = 4; x <= w - 6; x += spacing) {
    c.rect(x, 8, 2, h - 14, 'B');
    c.vline(x, 8, h - 14, 'w');
    c.vline(x + 1, 8, h - 14, 'k');
  }

  // Horizontal scaffold plank platforms
  const platformY1 = Math.floor(h * 0.35);
  const platformY2 = Math.floor(h * 0.65);

  c.rect(2, platformY1, w - 4, 3, 'W');
  c.rect(2, platformY1 + 1, w - 4, 2, 'w');
  c.hline(2, platformY1, w - 4, 'k');
  c.hline(2, platformY1 + 3, w - 4, 'k');

  c.rect(2, platformY2, w - 4, 3, 'W');
  c.rect(2, platformY2 + 1, w - 4, 2, 'w');
  c.hline(2, platformY2, w - 4, 'k');
  c.hline(2, platformY2 + 3, w - 4, 'k');

  // Diagonal cross bracing
  c.line(4, platformY1, w - 6, platformY2, 'b');
  c.line(w - 6, platformY1, 4, platformY2, 'b');

  // Wooden Ladder
  const lx = Math.floor(w * 0.7);
  c.vline(lx, platformY1 - 4, h - platformY1 - 2, 'B');
  c.vline(lx + 4, platformY1 - 4, h - platformY1 - 2, 'B');
  for (let ly = platformY1 - 2; ly < h - 8; ly += 5) {
    c.hline(lx, ly, 5, 'W');
  }

  // Props: pile of cut stone and timber planks at ground
  c.rect(6, h - 10, 6, 4, 'L');
  c.box(6, h - 10, 6, 4, 'k');
  c.rect(14, h - 9, 8, 3, 'W');
  c.box(14, h - 9, 8, 3, 'k');
}
