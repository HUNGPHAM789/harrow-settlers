// Generic helper to draw scaffolding over a building region
function drawGenericScaffold(c, x, y, w, h, topY = 16) {
  // Base footprint wall foundation
  c.rect(x + 2, y + h - 12, w - 4, 11, "s");
  c.rect(x + 2, y + h - 12, w - 4, 2, "E");
  c.box(x + 2, y + h - 12, w - 4, 11, "k");

  // Vertical timber uprights
  const p1 = x + 4;
  const p2 = x + Math.floor(w / 2);
  const p3 = x + w - 5;
  [p1, p2, p3].forEach((px) => {
    c.vline(px, topY + 2, h - topY - 4, "u");
    c.vline(px + 1, topY + 2, h - topY - 4, "U");
    c.px(px, topY + 2, "L");
  });

  // Horizontal ledger beams & plank floor
  const midY = topY + Math.floor((h - topY) / 2);
  [topY + 6, midY, y + h - 14].forEach((ly) => {
    c.hline(x + 2, ly, w - 4, "u");
    c.hline(x + 2, ly - 1, w - 4, "L");
  });

  // Cross braces
  c.line(p1, topY + 8, p2, midY, "w");
  c.line(p2, midY, p3, topY + 8, "w");
  c.line(p1, midY, p2, y + h - 14, "w");
  c.line(p2, y + h - 14, p3, midY, "w");

  // Construction ladder
  const lx = p2 + 3;
  if (lx + 5 < x + w - 4) {
    c.vline(lx, topY + 4, h - topY - 6, "W");
    c.vline(lx + 3, topY + 4, h - topY - 6, "W");
    for (let ry = topY + 8; ry < y + h - 6; ry += 5) {
      c.hline(lx, ry, 4, "L");
    }
  }

  // Stone block pile & wood planks on ground
  c.rect(x + 4, y + h - 6, 7, 5, "e");
  c.box(x + 4, y + h - 6, 7, 5, "k");
  c.px(x + 5, y + h - 5, "E");

  c.rect(x + w - 11, y + h - 5, 7, 4, "U");
  c.box(x + w - 11, y + h - 5, 7, 4, "k");
  c.hline(x + w - 10, y + h - 4, 5, "L");
}
